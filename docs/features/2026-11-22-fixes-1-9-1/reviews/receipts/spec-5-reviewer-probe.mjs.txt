// Spec 5 reviewer probe: replays the docsOnlySince path rule from payload/.joycraft/runtime/delivery.mjs.
import { readFileSync } from 'node:fs';
const src = readFileSync('payload/.joycraft/runtime/delivery.mjs', 'utf8');
const manifest = new RegExp(...(() => { const m = src.match(/const manifest = \/(.*)\/(\w*);/); return [m[1], m[2]]; })());
const docs = 'docs', records = ['docs/features', 'docs/archive'];
const isRecordPath = (path) => {
  if (!path.startsWith(`${docs}/`)) return false;
  if (manifest.test(path.split('/').at(-1))) return false;
  if (records.some(f => path.startsWith(`${f}/`))) return /\.(?:md|json|log|txt|diff|patch|ya?ml)$/i.test(path);
  return /\.(?:md|json)$/i.test(path);
};
const cases = [

  'docs/features/f/bun.lock', 'docs/features/f/bun.lockb', 'docs/features/f/deno.lock', 'docs/features/f/yarn.lock',
  'docs/archive/f/npm-shrinkwrap.json', 'docs/features/f/probe.mjs', 'docs/features/f/probe.mjs.txt',
  'docs/features/f/tsconfig.json.txt', 'docs/features/f/notes.md', 'docs/features/f/.yarnrc.yml',
];
